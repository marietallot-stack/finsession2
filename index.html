<script>

let score = 0;

let answered = 0;

const total = 3;


function check(button, correct) {

  if (button.classList.contains("clicked")) return;


  button.classList.add("clicked");

  answered++;


  if (correct) {

    button.classList.add("correct");

    score++;

  } else {

    button.classList.add("wrong");

  }


  if (answered === total) {

    let message = "";


    if (score === total) {

      message = "Excellent 🔥 Tu es prête à structurer ton positionnement.";

    } else if (score >= 2) {

      message = "Très bon niveau 👍 Il reste à affiner.";

    } else {

      message = "Base à consolider → retravailler le positionnement.";

    }


    document.getElementById("result").innerText =

      "Score : " + score + " / " + total + " — " + message;

  }

}

</script>
